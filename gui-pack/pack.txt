    ae_soc_pt.yml
    ...
  styles/
    tfl_standard.yml

  series.yml
  templates/...

template: orr_subgroup
label: Response rate with its CI, overall and in each subgroup level

  # by:         (required, no default)

    from: ADSL
    population: ITT                    # template's own: response analyses use ITT

    from: ADRS
    subjects_in: adsl_sub
    add: ["{by}"]                      # the subgroup columns: the report's add:
    where: !r PARAMCD == "BOR"
    keep: ["{category}"]
analysis:
  - id: BIGN
    label: Subjects per group
    method: categorical
    data: adsl_sub
    variables: ["{by}"]
    formats: "{formats}"
  - id: ORR
    label: Response rate with its CI, overall and in each subgroup level
    method: proportion_ci
    data: adrs_sub
    params: [subgroup, subgroup_lvl, where]
    conditions:                        # the ALL row first, then the report's rows (canon 782da95)
      - {subgroup: ALL, subgroup_lvl: ALL, where: !r TRUE}   # overall_row: true
      - "{conditions}"
    where: !r "{where}"
    derive: !r subgroup = "{subgroup}" | subgroup_lvl = "{subgroup_lvl}" | {category} = dplyr::if_else({category} %in% "{responder}", "{responder}", "Non-response")
    by: [subgroup, subgroup_lvl, "{by}"]
    variables: ["{category}"]
    args: !r value = list({category} = "{responder}"), method = "{ci_method}", conf.level = {conf_level}
    post: !r dplyr::filter(stat_name %in% c("N", "n", "estimate", "conf.low", "conf.high"))
    formats: "{formats}"
